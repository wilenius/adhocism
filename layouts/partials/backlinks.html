And the reference is matched on the *ref target*, never on the bare filename: a bare "id" matches any content containing "sidenotes", "video", "consider" … */ -}} {{- $currentTitle := .Title -}} {{- $currentFile := "" -}} {{- with .File }}{{ $currentFile = .BaseFileName }}{{ end -}} {{- $rxEscape := `([.\\+*?()|\[\]{}^$])` -}} {{- $titleRx := replaceRE $rxEscape `\$1` $currentTitle -}} {{- /* Body-reference patterns. */ -}} {{- $patterns := slice -}} {{- $patterns = $patterns | append (printf `#\[%s\]\(` $titleRx) -}} {{- $patterns = $patterns | append (printf `\[%s\]\(` $titleRx) -}} {{- $patterns = $patterns | append (printf `#%s\b` $titleRx) -}} {{- if $currentFile -}} {{- $fileRx := replaceRE $rxEscape `\$1` $currentFile -}} {{- $patterns = $patterns | append (printf `ref\s+"/pages/%s"` $fileRx) -}} {{- end -}} {{- $backlinks := slice -}} {{- range where .Site.RegularPages "Section" "pages" -}} {{- if eq .Title $currentTitle -}} {{- continue -}} {{- end -}} {{- $linked := false -}} {{- range .Params.tags -}} {{- if and .
