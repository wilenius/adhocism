{{/* Renders tag pills for a page. Each pill links to its tag page — either the Logseq page of that name, or the stub generated under content/tags/. Params: .page — the Hugo page .limit — 0 = all tags; N = show only first N */}} {{- $page := .page }} {{- $limit := .limit | default 0 }} {{/* Collect frontmatter tags, dropping empty strings (YAML comment artefacts) and malformed blobs — Logseq sometimes exports a whole tag list as one value, e.g. `#{"virkkaus" "fi"}`. scripts/generate_tag_pages.py applies the same filter, so a dropped tag never gets a page either. */}} {{- $valid := `^[\p{L}\p{N}][\p{L}\p{N} &'’/+._-]*$` }} {{- $rawTags := $page.Params.tags | default slice }} {{- $tags := slice }} {{- range $rawTags }} {{- if and . (findRE $valid .) }}{{- $tags = $tags | append . }}{{- end }} {{- end }} {{/* Fallback: first inline #hashtag from raw content when no property tags */}} {{- if eq (len $tags) 0 }} {{- $found := findRE `#([A-Za-z][A-Za-z0-9_]*)` $page.RawContent 1 }} {{- if $found }} {{- $tags = slice (strings.TrimPrefix "#" (index $found 0)) }} {{- end }} {{- end }} {{/* Trim to limit when limit > 0 */}} {{- if and (gt $limit 0) (gt (len $tags) $limit) }} {{- $tags = first $limit $tags }} {{- end }} {{- $index := partialCached "tag_index.html" . }} {{- if $tags }} {{- range $tags }} {{- $colorIndex := mod (index (sha256 .) 0) 8 }} {{- $href := index $index (. | lower) }} {{- if $href }} {{ . }} {{- else }} {{ . }} {{- end }} {{- end }} {{- end }}
