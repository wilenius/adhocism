%%
%% Pandoc + Tufte sidenote template
%% Renders footnotes as numbered sidenotes, [note: ...] as margin notes
%%
%% Column widths are set by the caller via YAML variables:
%%   textwidth_mm:   main text column width in mm  (default 130)
%%   marginwidth_mm: sidenote column width in mm   (default 35)
%% The script computes these from content length so sidenotes don't overflow.
%%
\documentclass[$if(fontsize)$$fontsize$,$endif$$if(lang)$lang=$lang$,$endif$$if(papersize)$$papersize$,$endif$a4paper]{tufte-handout}

% --- A4 geometry: caller controls text/margin split ---
% Total usable = A4 210mm - left 20mm - right 20mm - sep 5mm = 165mm
\geometry{
  a4paper,
  left=20mm,
  top=25mm,
  bottom=25mm,
  textwidth=$if(textwidth_mm)$$textwidth_mm$mm$else$130mm$endif$,
  marginparsep=5mm,
  marginparwidth=$if(marginwidth_mm)$$marginwidth_mm$mm$else$35mm$endif$
}

% morefloats extends the marginpar queue so packed sidenotes don't error.
\usepackage{morefloats}
% Explicitly re-bind \footnote -> \sidenote (tufte-handout does this too, but its
% \optparams form interacts badly with morefloats at wide margin widths).
\renewcommand{\footnote}[1]{\sidenote{#1}}

% --- [note: ...] support via Lua filter ---
% Unnumbered margin notes are inserted by note-filter.lua as \marginnote{...}

% --- Standard pandoc template pieces ---
$if(highlighting-macros)$
$highlighting-macros$
$endif$

$for(header-includes)$
$header-includes$
$endfor$

% --- Hyperref ---
\hypersetup{%
  $if(title)$pdftitle={$title$},$endif$%
  $if(author)$pdfauthor={$author$},$endif$%
  colorlinks=true,
  linkcolor=$if(linkcolor)$$linkcolor$$else$blue$endif$,
  urlcolor=$if(urlcolor)$$urlcolor$$else$blue$endif$,
  citecolor=$if(citecolor)$$citecolor$$else$blue$endif$%
}

$if(csl-refs)$
\newlength{\cslhangindent}
\setlength{\cslhangindent}{1.5em}
\newlength{\csllabelwidth}
\setlength{\csllabelwidth}{3em}
\newenvironment{CSLReferences}[2]%
  {$if(csl-hanging-indent)$\setlength{\parindent}{0pt}%
   \everypar{\setlength{\hangindent}{\cslhangindent}}\ignorespaces$endif$}%
  {\par}
$endif$

\title{$title$}
$if(subtitle)$\subtitle{$subtitle$}$endif$
\author{$for(author)$$author$$sep$ \and $endfor$}
$if(date)$\date{$date$}$endif$

\begin{document}

$if(title)$
\maketitle
$endif$

$if(abstract)$
\begin{abstract}
\noindent
$abstract$
\end{abstract}
$endif$

$if(toc)$
\tableofcontents
$endif$

$body$

% Stretch the main column to fill the last page so any overflow sidenotes
% on a subsequent page don't appear beside an empty left column.
\vspace*{\fill}
\clearpage

\end{document}
